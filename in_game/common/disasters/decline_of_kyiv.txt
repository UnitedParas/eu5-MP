decline_of_kyiv = {
    image = "gfx/interface/illustrations/disasters/decline_of_kyiv.dds"
    monthly_spawn_chance = monthly_spawn_chance_unique
    fire_only_once = yes

    can_start = {
        tag = KIE
        has_any_active_disaster = no
		has_ruler = yes
    }

    can_end = {
        in_civil_war = no
        OR = { 
            var:kie_lithuanian_influence_var >= 100
            var:kie_mongol_influence_var >= 100
        }
    }

	modifier = {
		global_army_levy_size_modifier = -0.2
		tax_income_efficiency = -0.2
		global_monthly_control = -0.005
		global_max_control = -0.15
	}

	on_start = {
		set_variable = { name = kie_lithuanian_influence_var value = 0 }
		set_variable = { name = kie_mongol_influence_var value = 0 }
		trigger_event_non_silently = decline_of_kyiv.1
	}

	on_end = {
		if = {
			limit = {kie_lithuanian_influence_var >= 100}
			trigger_event_non_silently = decline_of_kyiv.2
		}
		else_if = {
			limit = {kie_mongol_influence_var >= 100}
			trigger_event_non_silently = decline_of_kyiv.3
		}
		else = {
			trigger_event_non_silently = decline_of_kyiv.4
		}
	}

	on_monthly = {
		if = {
			limit = {
				has_variable = kie_lithuanian_influence_var
				has_variable = kie_mongol_influence_var
			}
			random_list = {
				20 = {custom_tooltip = kyivan_event_occurs_tt}
				80 = {custom_tooltip = no_event_occurs_tt}
			}
			hidden_effect = {
				random_list = {
					80 = { }
					2 = {
						trigger_event_silently = decline_of_kyiv.4
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.5
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.6
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.7
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.8
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.9
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.10
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.11
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.12
					}
					2 = {
						trigger_event_silently = decline_of_kyiv.13
					}
				}
			}
		}
	}
}
