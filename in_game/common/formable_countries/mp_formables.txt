		midlands_area
		northumbria_area
		east_anglia_area
		west_country_area
		wales_area
		leinster_area
		normandy_area
		anjou_area
		brittany_area
		poitou_area
		guyenne_area
		gascony_area
		limousin_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_anv_formed

		}
		# Empire rank, not kingdom. Level 3 formables in this mod split
		# roughly evenly between the two (10 empire / 15 kingdom), so the
		# deciding factor is scope, and this one requires England, Wales, a

		# 14 areas: a larger realm than several of the level-3 formables
		# that already grant empire (Latin Empire, Tibetan Empire). It is
		# also the only formable in the mod whose *name* is an empire while
		# its holder stayed a king, which players noticed and asked about
		# (Workshop, 2026-07-22).
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
	}
}

IRQ_f = {

	level = 3
	required_locations_fraction = 0.5
	rule = plausible

	# Iraq is founded from Iraq, never annexed as a distant province.
	# Same idiom vanilla uses for USA_f and CAN_f.
	capital_required = yes

	potential = {
		religion.group = religion_group:muslim
		OR = {
			# The Arab claim: Iraqi, Bedouin, Hijazi, Levantine and the rest of arabic_group.
			culture = { has_culture_group = culture_group:arabic_group }
			# The Baghdad claim: whoever actually rules Mesopotamia from inside it.
			# This is the branch that keeps the Jalayirids (mongolian_culture, capital
			# at Baghdad in 1337) eligible, and it is also open to the Qara Qoyunlu and
			# Aq Qoyunlu once they move their seat to the Tigris, as Jahan Shah did.
			capital = { area = area:iraq_arabi_area }
		}
	}

	allow = {
		owns = location:baghdad
		owns = location:basra
		owns = location:kufa
		in_civil_war = no
	}

	name = IRQ
	flag = IRQ
	adjective = IRQ_ADJ
	tag = IRQ
	color = map_IRQ

	areas = {
		iraq_arabi_area
		jazira_area
	}

	form_effect = {
		set_capital = location:baghdad
		# The state adopts the identity of the land it rules. The proposal's
		# "Iraqi culture" is an outcome of formation, not a gate on it, because
		# no country on the 1337 map starts with iraqi_culture as its primary.
		# Pops are deliberately left alone: Mesopotamia stayed plural.
		if = {
			limit = { NOT = { culture = culture:iraqi_culture } }
			change_culture = culture:iraqi_culture
		}
		# Sultanate of Iraq, the Jalayirid rank restored.
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
		# The formed markers are stamped here rather than only in the
		# formation event's options, so the content layer never depends on a
		# popup having been delivered. Hidden because the form_effect tooltip
		# should read as history, not as bookkeeping. The options set these
		# again, which is harmless: set_variable is idempotent. The path
		# variable stays in the event, because it is the player's answer.
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_irq_formed
		}
	}
}

OUT_f = {

	level = 4
	required_locations_fraction = 0.6
	rule = plausible

	potential = {
		religion = religion:catholic

		# The Latin East, not any Catholic realm that reaches the Jordan. The
		# five crusader formables are named rather than left to the rank
		# clause because Crusader Africa is the one that does not raise rank
		# on formation and would otherwise fall through.
		OR = {
			has_or_had_tag = KOJ
			has_or_had_tag = LAT
			has_or_had_tag = KOA
			has_or_had_tag = TKT
			has_or_had_tag = HOS
			has_reform = government_reform:military_order_reform
			country_rank = country_rank:rank_kingdom
			country_rank = country_rank:rank_empire
		}
	}

	allow = {
		# The vow and the seat. Jerusalem is definitional and the Workshop
		# proposal omitted it; Acre was the kingdom's actual capital from 1191
		# and its last mainland holding in 1291.
		owns = location:jerusalem
		owns = location:acre

		# The Egyptian project, the Mesopotamian dividend of the Mongol
		# alliance, the Hejaz that Raynald sailed for, and the Red Sea mouth.
		owns = location:cairo
		owns = location:baghdad
		owns = location:mosul
		owns = location:mecca
		owns = location:medina
		owns = location:adan

		in_civil_war = no

		# The governance clause, and the point of the whole country. Outremer
		# was a Frankish ruling caste over an overwhelmingly non-Latin
		# population, so a realm that has converted or emptied the East into a
		# Catholic majority is not Outremer, it is Jerusalem written large.
		# This also excludes a European superpower that keeps the Middle East
		# as an appendage: France with a Levantine province is France. The
		# country is named for being overseas, and this is the line that
		# enforces it.
		religion_percentage_in_country = {
			religion = religion:catholic
			value <= 0.5
		}
	}

	name = OUT
	flag = OUT
	adjective = OUT_ADJ
	tag = OUT
	color = map_OUT

	regions = {
		crescent_region
		egypt_region
	}
	areas = {
		hedjaz_area
		south_yemen_area
	}
	provinces = {
		# The seat of the exiled kingdom after 1291 and the staging base for
		# Peter I's Alexandria crusade of 1365.
		cyprus_province
	}

	form_effect = {

		# Formation markers live HERE, not only in the event's options.

		# A country can be formed by script (vanilla forms JAP from the

		# sengoku situation), and then the popup may never be answered,

		# leaving every advance gated on this variable unreachable.

		hidden_effect = {

			set_variable = nd_formed

			set_variable = nd_out_formed

		}
		if = {
			limit = { NOT = { capital = location:jerusalem } }
			set_capital = location:jerusalem
		}
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
	}
}
