placate_the_lithuanians = {
	type = disaster

	potential = { 
		country_exists = c:LIT 
		c:LIT = {
			OR = {
				is_subject = no
				is_subject_of = scope:actor
			}
		}
	}

	allow = {
		scope:actor = { 
			NOR = {
				is_rival_of = c:LIT
				is_enemy_of = c:LIT
				is_at_war_with = c:LIT
			}
		}

		gold > {
				value = monthly_income_total
				multiply = 4
			}
	}

	ai_tick = monthly
	ai_tick_frequency = 4
	price = price:fate_of_phoenix_actions_price

	cooldown = {
		type = placate_the_lithuanians
		years = 2
	}
	
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:decline_of_kyiv
			disaster_is_active = yes
		}
	}

	effect = {
		c:LIT = { trigger_event_non_silently = decline_of_kyiv.16 }

		custom_tooltip = placate_the_lithuanians_tt
	}

	ai_will_do = {
		if = {
			limit = { 
				scope:actor.government_power > 90
			}

			add = {
				value = { 1 50 }
				desc = "high government power"
			}
		}
		else = {
			subtract = {
				value = 1
				desc = "low government power"
			}
		}
	}
}
