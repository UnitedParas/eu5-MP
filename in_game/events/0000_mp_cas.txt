namespace = flavor_cas

flavor_cas.2200 = { #The Iberian Wedding
	hide_portraits = yes
	fire_only_once = yes
	title = flavor_cas.2200.title
	desc = flavor_cas.2200.desc

	fire_only_once = yes

	dynamic_historical_event = {
		tag = CAS
		tag = SPA
		from = 1444.1.1
		to = 1836.1.1
		monthly_chance = 5
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	trigger = {
		is_ai = yes
		c:ARA = { is_ai = yes }
		not = { in_union_with = c:ARA }
		or = {
			and = {
				ruler ?= { is_female = yes is_regent = no }
				c:ARA = { ruler ?= { is_female = no is_regent = no } }
			}
			and = {
				ruler ?= { is_female = no is_regent = no }
				c:ARA = { ruler ?= { is_female = yes is_regent = no } }
			}
		}
		ruler ?= { save_scope_as = castile_ruler }
	}

	option = { #PU
		name = ht_ai_only.a
		historical_option = yes
		c:ARA = {
			set_new_ruler = scope:castile_ruler
			make_subject_of = {
				target = root
				type = subject_type:fiefdom
			}
		}
		ai_chance = {
			factor = 75
		}
	}
	option = { #No PU
		name = ht_ai_only.b
		ai_chance = {
			factor = 25
		}
	}
}
